cmake_minimum_required(VERSION 3.27)
project(CyCLAPBridge VERSION 0.2.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(Threads REQUIRED)
option(CYCLAPBRIDGE_BUILD_CLI "Build the conversion command line application" ON)
option(CYCLAPBRIDGE_BUILD_GUI "Build the native Windows conversion application" ON)
option(CYCLAPBRIDGE_BUILD_TESTS "Build original fixtures and native tests" OFF)
option(CYCLAPBRIDGE_INSTALL_BACKEND "Install pinned wrapper sources and SDKs for offline conversion" ON)

add_library(CyCLAPBridge STATIC src/Conversion.cpp src/Process.cpp src/Plugin.cpp
  src/HostedPlugin.cpp src/Lv2Host.cpp src/JsfxHost.cpp src/Vst3Host.cpp src/MultiConversion.cpp src/EditorIids.cpp)
add_library(CyCLAPBridge::CyCLAPBridge ALIAS CyCLAPBridge)
target_link_libraries(CyCLAPBridge PRIVATE Threads::Threads)
target_compile_features(CyCLAPBridge PUBLIC cxx_std_20)
set_target_properties(CyCLAPBridge PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
include(cmake/Runtimes.cmake)
target_include_directories(CyCLAPBridge PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
  PRIVATE third_party/clap/include)
if(WIN32)
  target_link_libraries(CyCLAPBridge PRIVATE user32 gdi32 comdlg32 shell32 msimg32)
  target_compile_definitions(CyCLAPBridge PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
elseif(APPLE)
  enable_language(OBJCXX)
  set_target_properties(CyCLAPBridge PROPERTIES OBJCXX_STANDARD 20 OBJCXX_STANDARD_REQUIRED ON)
  target_sources(CyCLAPBridge PRIVATE src/AuHost.cpp src/AuEditor.mm)
  set_source_files_properties(src/AuEditor.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
  target_link_libraries(CyCLAPBridge PRIVATE "-framework CoreFoundation" "-framework AudioToolbox" "-framework AudioUnit" "-framework Cocoa")
else()
  target_link_libraries(CyCLAPBridge PRIVATE ${CMAKE_DL_LIBS})
endif()
if(MSVC)
  target_compile_options(CyCLAPBridge PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(CyCLAPBridge PRIVATE -Wall -Wextra -Wpedantic)
endif()

function(cybridge_application target)
  target_link_libraries(${target} PRIVATE CyCLAPBridge::CyCLAPBridge)
  target_include_directories(${target} PRIVATE apps)
  target_compile_definitions(${target} PRIVATE
    CYBRIDGE_SOURCE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}"
    CYBRIDGE_CMAKE_EXECUTABLE="${CMAKE_COMMAND}")
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /utf-8)
  endif()
  install(TARGETS ${target} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endfunction()
if(CYCLAPBRIDGE_BUILD_CLI)
  add_executable(cyclapbridge-cli apps/cli.cpp)
  cybridge_application(cyclapbridge-cli)
endif()
if(CYCLAPBRIDGE_BUILD_GUI AND WIN32)
  add_executable(CyCLAPBridgeStudio WIN32 apps/windows/Studio.cpp apps/windows/Studio.manifest)
  cybridge_application(CyCLAPBridgeStudio)
  target_compile_definitions(CyCLAPBridgeStudio PRIVATE NOMINMAX UNICODE _UNICODE WIN32_LEAN_AND_MEAN)
  target_link_libraries(CyCLAPBridgeStudio PRIVATE comctl32 comdlg32 shell32 ole32 shlwapi)
  if(MSVC)
    target_link_options(CyCLAPBridgeStudio PRIVATE "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/apps/windows/Studio.manifest")
  endif()
endif()

install(TARGETS CyCLAPBridge EXPORT CyCLAPBridgeTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT CyCLAPBridgeTargets NAMESPACE CyCLAPBridge:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/CyCLAPBridge)

  "${CMAKE_CURRENT_BINARY_DIR}/CyCLAPBridgeConfig.cmake"
  INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/CyCLAPBridge)

  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/CyCLAPBridgeConfig.cmake"
  "${CMAKE_CURRENT_BINARY_DIR}/CyCLAPBridgeConfigVersion.cmake"
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/CyCLAPBridge)
